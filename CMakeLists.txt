cmake_minimum_required(VERSION 3.24)
project(chr_dawn LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "The Dawn experiment requires Clang.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
find_package(Vulkan REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
find_package(Dawn CONFIG REQUIRED)
file(READ native/scene.wgsl CHR_SHADER)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS native/scene.wgsl)
configure_file(native/shader.hpp.in generated/shader.hpp @ONLY)
add_executable(chr-dawn native/main.cpp native/media.cpp native/bridge.cpp native/renderer.cpp native/scene.cpp)
target_include_directories(chr-dawn PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(chr-dawn PRIVATE dawn::webgpu_dawn PkgConfig::FFMPEG Vulkan::Vulkan nlohmann_json::nlohmann_json)
install(TARGETS chr-dawn RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_executable(chr-scene-tests native/tests.cpp native/scene.cpp)
  target_link_libraries(chr-scene-tests PRIVATE PkgConfig::FFMPEG nlohmann_json::nlohmann_json)
  add_test(NAME scene-identity COMMAND chr-scene-tests)
endif()
